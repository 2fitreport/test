                                                        fontSize: '12px',
                                                        overflow: 'hidden',

                                                        color: '#666',
                                                        fontWeight: '500'
                                                    }}>
                                                        {cretopFile?.name}

                                    handleViewFile(nextFile);
                                    setCurrentFileIndex(nextIndex);
